How to Download Generated Images

Here's the JavaScript code to download an image from a URL. You can use this in your own projects!

download.js 

async function downloadImage(imageUrl, filename = 'ai-image.png') {
  try {
    // 1. Fetch the image data from the URL
    const response = await fetch(imageUrl);
    if (!response.ok) {
        throw new Error(`Network response was not ok: ${response.statusText}`);
    }
    const blob = await response.blob();

    // 2. Create a temporary URL for the blob data
    const url = window.URL.createObjectURL(blob);

    // 3. Create a temporary anchor element to trigger the download
    const link = document.createElement('a');
    link.href = url;
    link.download = filename;

    // 4. Append to the document, click, and then remove
    document.body.appendChild(link);
    link.click();
    document.body.removeChild(link);

    // 5. Clean up by revoking the temporary URL
    window.URL.revokeObjectURL(url);
  } catch (error) {
    console.error('Failed to download the image:', error);
    alert('Could not download the image. See console for details.');
  }
}
